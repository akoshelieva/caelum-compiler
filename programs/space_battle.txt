astro Ship ⟨
    flow ☼ hp.
    flow ☼ energy.
    flow ☼ heat.
    flow ☼ shield.
    flow ☼ charge.

    spark takeHit(☼ damage) -> ☼ ⟨
        hp = hp - damage.
        when (14 - hp) / 14 != 0 ⟨
            hp = 0.
        ⟩
        rise hp.
    ⟩
⟩

flow Ship player = Ship⟨hp = 14, energy = 7, heat = 0, shield = 0, charge = 0⟩.
flow Ship enemy = Ship⟨hp = 14, energy = 0, heat = 0, shield = 0, charge = 0⟩.
flow ☼ fireStreak = 0.
flow ☼ defendStreak = 0.
flow ☼ empWindow = 0.
flow ☼ enemyRage = 0.
flow ☁ finished = dark.

spark firePower(☼ power) -> ☼ ⟨
    rise power + 1.
⟩

spark shieldImpact(☼ damage) -> ☼ ⟨
    when empWindow == 1 ⟨
        rise damage + 3.
    ⟩
    when enemy ~ shield == 1 ⟨
        rise damage - 3.
    ⟩
    rise damage.
⟩

spark showField(Ship ally, Ship rival) -> ☼ ⟨
    recordCoordinates("+---------------------------+").
    recordCoordinates("|   PLAYER        ENEMY     |").
    recordCoordinates("|    [^]           <=>      |").
    recordCoordinates("|    /|\\            |       |").
    recordCoordinates("+---------------------------+").
    recordCoordinates("PLAYER HP:").
    recordCoordinates(ally ~ hp).
    recordCoordinates("ENERGY:").
    recordCoordinates(ally ~ energy).
    recordCoordinates("REACTOR HEAT:").
    recordCoordinates(ally ~ heat).
    recordCoordinates("ENEMY HP:").
    recordCoordinates(rival ~ hp).
    recordCoordinates("ENEMY SHIELD (0/1):").
    recordCoordinates(rival ~ shield).
    recordCoordinates("ENEMY LASER (0/1):").
    recordCoordinates(rival ~ charge).
    rise 0.
⟩

spark playTurn(☼ action, ☼ turn) -> ☼ ⟨
    flow ☼ guarded = 0.
    flow ☼ chargedAtStart = enemy ~ charge.

    when action != 1 ⟨
        when action != 2 ⟨
            when action != 3 ⟨
                recordCoordinates("INVALID ACTION: Choose 1, 2 or 3.").
                finished = light.
            ⟩
        ⟩
    ⟩
    when action == 1 ⟨
        when player ~ energy / 2 == 0 ⟨
            recordCoordinates("INVALID ACTION: Not enough energy for Fire.").
            finished = light.
        ⟩
    ⟩
    when action == 3 ⟨
        when player ~ energy / 3 == 0 ⟨
            recordCoordinates("INVALID ACTION: Not enough energy for EMP.").
            finished = light.
        ⟩
    ⟩

    when finished == dark ⟨
        when action == 1 ⟨
            player ~ energy = player ~ energy - 2.
            player ~ heat = player ~ heat + 3.
            fireStreak = fireStreak + 1.
            defendStreak = 0.
            flow ☼ damage = 4 |> firePower |> shieldImpact.
            enemy ~ takeHit(damage).
            recordCoordinates("FIRE: Damage:").
            recordCoordinates(damage).
            empWindow = 0.
            when fireStreak == 2 ⟨
                when enemy ~ hp != 0 ⟨
                    enemy ~ shield = 1.
                    recordCoordinates("ENEMY: Shield activated.").
                ⟩
            ⟩
        ⟩ otherwise ⟨
            when action == 2 ⟨
                guarded = 1.
                player ~ energy = player ~ energy + 4.
                when player ~ energy / 9 != 0 ⟨
                    player ~ energy = 8.
                ⟩
                player ~ heat = player ~ heat - 4.
                when (8 - player ~ heat) / 8 != 0 ⟨
                    player ~ heat = 0.
                ⟩
                fireStreak = 0.
                defendStreak = defendStreak + 1.
                recordCoordinates("DEFEND: Energy +4, heat -4.").
                when defendStreak == 2 ⟨
                    enemy ~ charge = 1.
                    recordCoordinates("ENEMY: Charging heavy laser.").
                ⟩
            ⟩ otherwise ⟨
                player ~ energy = player ~ energy - 3.
                player ~ heat = player ~ heat + 1.
                enemy ~ shield = 0.
                enemy ~ takeHit(1).
                empWindow = 1.
                enemyRage = 2.
                fireStreak = 0.
                defendStreak = 0.
                recordCoordinates("EMP: Shield disabled, next Fire +3.").
                recordCoordinates("ENEMY: Aggressive response.").
            ⟩
        ⟩

        when player ~ heat / 8 != 0 ⟨
            recordCoordinates("DEFEAT: Reactor meltdown.").
            finished = light.
        ⟩

        when finished == dark ⟨
            when enemy ~ hp == 0 ⟨
                when (14 - player ~ hp) / 10 != 0 ⟨
                    recordCoordinates("VICTORY: Barely survived!").
                ⟩ otherwise ⟨
                    recordCoordinates("VICTORY: Enemy destroyed!").
                ⟩
                finished = light.
            ⟩ otherwise ⟨
                flow ☼ retaliation = 2.
                when enemyRage != 0 ⟨
                    retaliation = retaliation + 2.
                    enemyRage = enemyRage - 1.
                ⟩
                when chargedAtStart == 1 ⟨
                    retaliation = retaliation + 4.
                    enemy ~ charge = 0.
                    recordCoordinates("ENEMY: Heavy laser fired!").
                ⟩
                when turn == 5 ⟨
                    retaliation = retaliation + 5.
                    recordCoordinates("ENEMY: Final strike!").
                ⟩ otherwise ⟨
                    recordCoordinates("ENEMY: Return fire.").
                ⟩
                when guarded == 1 ⟨
                    retaliation = retaliation - 3.
                    when retaliation == -1 ⟨
                        retaliation = 0.
                    ⟩
                ⟩
                player ~ takeHit(retaliation).
                recordCoordinates("DAMAGE TAKEN:").
                recordCoordinates(retaliation).
                when player ~ hp == 0 ⟨
                    recordCoordinates("DEFEAT: Ship destroyed.").
                    finished = light.
                ⟩
            ⟩
        ⟩
    ⟩
    showField(player, enemy).
    when finished == dark ⟨
        when turn == 5 ⟨
            recordCoordinates("DEFEAT: Enemy survived five rounds.").
            finished = light.
        ⟩
    ⟩
    rise 0.
⟩
